\begin{abstract}
This document formulates the mathematical properties, probability normalization mechanics, and numerical stabilization techniques for the Softmax operator in classification and attention mechanisms. Every concept is developed across three levels of explanation: intuition, formal formulation, and practical application. We derive the exact shift-invariance identity, formulate the log-sum-exp numerical safeguard, prove the exactness of single-pass online streaming softmax, state the temperature scaling entropy theorem with separated assumptions, guarantees, and limitations, and provide an exact verified numerical trace matching the production implementation contract.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbb{R}$ denote the set of real numbers, $\mathbb{R}^K$ the $K$-dimensional real vector space, and $\Delta^{K-1} = \{\mathbf{p} \in \mathbb{R}^K \mid p_k \ge 0, \sum_{k=1}^K p_k = 1\}$ the standard probability simplex. Matrices are denoted by uppercase bold letters ($\mathbf{Z}, \mathbf{P}$), vectors by lowercase bold letters ($\mathbf{z}, \mathbf{p}$), and scalars by standard italics ($z, T$).

\subsection{Softmax Normalization Operator}
\paragraph{Intuition:} The softmax function transforms an arbitrary vector of unconstrained real numbers (logits or energy scores) into a valid discrete probability distribution where every probability is strictly positive and the entire distribution sums to exactly $1.0$.

\paragraph{Mathematical Formulation:}
For a logit vector $\mathbf{z} = [z_1, z_2, \dots, z_K]^\top \in \mathbb{R}^K$ and temperature $T > 0$, the softmax operator $\mathcal{S}_T: \mathbb{R}^K \to \Delta^{K-1}$ is defined as:
\begin{equation}
\label{eq:softmax_def}
p_k = [\mathcal{S}_T(\mathbf{z})]_k = \frac{\exp(z_k / T)}{\sum_{j=1}^K \exp(z_j / T)}, \quad \forall k \in \{1, \dots, K\}.
\end{equation}

\paragraph{Worked Numerical Example:}
Let $\mathbf{z} = [1000.0, 1002.0, 1001.0]^\top$ and $T = 1.0$.
Direct exponentiation fails due to overflow ($e^{1000} \to \infty$).
Subtracting $m = \max(\mathbf{z}) = 1002.0$:
\begin{align}
\tilde{\mathbf{z}} &= [1000.0 - 1002.0, 1002.0 - 1002.0, 1001.0 - 1002.0]^\top = [-2.0, 0.0, -1.0]^\top, \\
\exp(\tilde{\mathbf{z}}) &= [e^{-2.0}, e^0, e^{-1.0}]^\top \approx [0.135335, 1.0, 0.367879]^\top, \\
\sum_{j=1}^3 \exp(\tilde{z}_j) &\approx 0.135335 + 1.0 + 0.367879 = 1.503214.
\end{align}
Normalized probabilities:
\begin{equation}
\mathbf{p} = \left[ \frac{0.135335}{1.503214}, \frac{1.0}{1.503214}, \frac{0.367879}{1.503214} \right]^\top \approx [0.090031, 0.665241, 0.244728]^\top.
\end{equation}

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size (number of independent distributions).
    \item $K \in \mathbb{N}$: Class vocabulary or sequence length dimension.
    \item $\mathbf{Z} \in \mathbb{R}^{B \times K}$: Input logit matrix.
    \item $\mathbf{P} \in \mathbb{R}^{B \times K}$: Output probability distribution matrix (or log-probabilities).
    \item $T \in \mathbb{R}_{>0}$: Temperature parameter scaling distribution sharpness.
    \item $\text{LSE}(\mathbf{z}) = \ln \left( \sum_{k=1}^K e^{z_k} \right)$: Exact Log-Sum-Exp scalar.
    \item $m = \max_{1 \le k \le K} z_k$: Maximum logit coordinate per row.
\end{itemize}

\section{Problem Definition and Core Properties}
\label{sec:problem_definition}

\subsection{Shift-Invariance Property}
\begin{property}[Exact Translation Invariance]
\label{prop:shift_inv}
For any logit vector $\mathbf{z} \in \mathbb{R}^K$ and arbitrary scalar constant $c \in \mathbb{R}$:
\begin{equation}
\mathcal{S}(\mathbf{z} - c \cdot \mathbf{1}_K) = \mathcal{S}(\mathbf{z}).
\end{equation}
\end{property}
\begin{proof}
For any coordinate $k \in \{1, \dots, K\}$:
\begin{equation}
\frac{\exp(z_k - c)}{\sum_{j=1}^K \exp(z_j - c)} = \frac{e^{-c} \exp(z_k)}{e^{-c} \sum_{j=1}^K \exp(z_j)} = \frac{\exp(z_k)}{\sum_{j=1}^K \exp(z_j)}.
\end{equation}
Setting $c = m = \max_{j} z_j$ guarantees that every shifted exponent satisfies $\tilde{z}_k = z_k - m \le 0$, ensuring $\exp(\tilde{z}_k) \in (0, 1]$ and strictly preventing floating-point overflow.
\end{proof}

\subsection{Log-Sum-Exp Trick}
\begin{definition}[Numerically Stable Log-Sum-Exp]
The log-sum-exp operator computes the log partition function:
\begin{equation}
\label{eq:lse}
\text{LSE}(\mathbf{z}) = \ln \left( \sum_{k=1}^K e^{z_k} \right) = m + \ln \left( \sum_{k=1}^K e^{z_k - m} \right), \quad \text{where } m = \max_{1 \le j \le K} z_j.
\end{equation}
The exact log-softmax is computed without intermediate exponentiation division:
\begin{equation}
\ln p_k = z_k - \text{LSE}(\mathbf{z}) = (z_k - m) - \ln \left( \sum_{j=1}^K e^{z_j - m} \right).
\end{equation}
\end{definition}

\section{Algorithmic Specification: Standard and Streaming Modes}
\label{sec:algorithm}

Algorithm~\ref{alg:softmax_stable} specifies both the two-pass stable algorithm and the one-pass online streaming algorithm (Milakov and Gimelshein 2018 / Dao et al. 2022).

\begin{algorithm}[H]
\caption{Numerically Stable Softmax and Online Streaming Evaluation}
\label{alg:softmax_stable}
\begin{algorithmic}[1]
\Require Logit batch $\mathbf{Z} \in \mathbb{R}^{B \times K}$, temperature $T > 0$, mode $\in \{\text{standard\_stable}, \text{log\_softmax}, \text{online\_streaming}\}$.
\Ensure Output matrix $\mathbf{P} \in \mathbb{R}^{B \times K}$, log-sum-exp vector $\mathbf{s}_{\text{lse}} \in \mathbb{R}^B$.
\State $B \gets \text{rows}(\mathbf{Z}), \quad K \gets \text{cols}(\mathbf{Z})$
\State $\mathbf{P} \gets \mathbf{0} \in \mathbb{R}^{B \times K}, \quad \mathbf{s}_{\text{lse}} \gets \mathbf{0} \in \mathbb{R}^B$
\For{$i = 1 \textbf{ to } B$}
    \If{$\text{mode} = \text{online\_streaming}$}
        \State $m \gets -\infty, \quad d \gets 0.0$
        \For{$k = 1 \textbf{ to } K$}
            \State $x \gets Z_{ik} / T$
            \If{$x > m$}
                \State $d \gets d \cdot \exp(m - x) + 1.0$
                \State $m \gets x$
            \Else
                \State $d \gets d + \exp(x - m)$
            \EndIf
        \EndFor
        \State $s_{\text{lse}} \gets m + \ln(d)$
        \For{$k = 1 \textbf{ to } K$}
            \State $P_{ik} \gets \exp(Z_{ik}/T - m) / d$
        \EndFor
    \Else
        \State $m \gets \max_{1 \le k \le K} (Z_{ik} / T)$
        \State $\text{sum\_exp} \gets \sum_{k=1}^K \exp(Z_{ik} / T - m)$
        \State $s_{\text{lse}} \gets m + \ln(\text{sum\_exp})$
        \For{$k = 1 \textbf{ to } K$}
            \If{$\text{mode} = \text{log\_softmax}$}
                \State $P_{ik} \gets (Z_{ik} / T - m) - \ln(\text{sum\_exp})$
            \Else
                \State $P_{ik} \gets \exp(Z_{ik} / T - m) / \text{sum\_exp}$
            \EndIf
        \EndFor
    \EndIf
    \State $s_{\text{lse}}^{(i)} \gets s_{\text{lse}}$
\EndFor
\State \Return $(\mathbf{P}, \mathbf{s}_{\text{lse}})$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Temperature Scaling and Streaming Equivalence}
\label{sec:correctness}

\begin{theorem}[Online Streaming Softmax Correctness]
\label{thm:online_softmax}
\textbf{Assumptions:}
Let logit sequence $(z_1, z_2, \dots, z_K)$ be processed element-by-element with running maximum $m_k = \max(m_{k-1}, z_k)$ and running normalized denominator $d_k = d_{k-1} e^{m_{k-1} - m_k} + e^{z_k - m_k}$.

\textbf{Guarantee:}
At step $K$, the state $(m_K, d_K)$ satisfies:
\begin{equation}
m_K = \max_{1 \le j \le K} z_j, \quad d_K = \sum_{j=1}^K e^{z_j - m_K}, \quad \text{and} \quad \frac{e^{z_k - m_K}}{d_K} = [\mathcal{S}(\mathbf{z})]_k.
\end{equation}
The streaming pass computes the exact numerically stable softmax without requiring a full upfront reduction pass.

\textbf{Limitations:}
Online streaming updates require updating the accumulator $d$ with exponential factor $e^{m_{k-1} - m_k}$ whenever a new maximum is found, requiring IEEE 754 float precision.
\end{theorem}

\begin{theorem}[Temperature Scaling Limits]
\label{thm:temperature}
For any distinct logit vector $\mathbf{z} \in \mathbb{R}^K$ with unique maximum coordinate $k^* = \arg\max_k z_k$:
\begin{align}
\lim_{T \to 0^+} [\mathcal{S}_T(\mathbf{z})]_k &= \begin{cases} 1 & \text{if } k = k^*, \\ 0 & \text{if } k \ne k^* \end{cases} \quad \text{(Argmax One-Hot Indicator)}, \\
\lim_{T \to \infty} [\mathcal{S}_T(\mathbf{z})]_k &= \frac{1}{K} \quad \text{(Maximum Entropy Uniform Distribution)}.
\end{align}
\end{theorem}

\section{Complexity Derivation}
\label{sec:complexity}

\subsection{Time Complexity}
\begin{enumerate}
    \item Standard two-pass stable mode: Finding maximum across row $i$ takes $K-1$ comparisons. Exponentiating and summing takes $K$ exponentials and $K-1$ additions. Normalizing takes $K$ divisions. Total operations per row: $3K$.
    \item For $B$ rows:
    \begin{equation}
    T(B, K) = \mathcal{O}(B \cdot K).
    \end{equation}
\end{enumerate}

\subsection{Space Complexity}
\begin{enumerate}
    \item Output matrix $\mathbf{P}$ allocates $B \cdot K$ floating-point scalars.
    \item Total auxiliary storage:
    \begin{equation}
    S(B, K) = \mathcal{O}(B \cdot K).
    \end{equation}
\end{enumerate}

\section{Worked Example and Numerical Trace}
\label{sec:worked_example}

Consider a batch of size $B = 1$ with $K = 3$ logits: $\mathbf{z} = [2.0, 1.0, 0.1]^\top$ and $T = 1.0$.

\subsection{Step-by-Step Stable Evaluation}
\begin{align}
m &= \max(2.0, 1.0, 0.1) = 2.0. \\
\tilde{\mathbf{z}} &= [2.0 - 2.0, 1.0 - 2.0, 0.1 - 2.0]^\top = [0.0, -1.0, -1.9]^\top. \\
\exp(\tilde{\mathbf{z}}) &= [1.0, e^{-1.0}, e^{-1.9}]^\top \approx [1.0, 0.367879, 0.149569]^\top. \\
\text{sum\_exp} &= 1.0 + 0.367879 + 0.149569 = 1.517448. \\
\text{LSE}(\mathbf{z}) &= 2.0 + \ln(1.517448) \approx 2.0 + 0.417031 = 2.417031.
\end{align}
Normalized probabilities:
\begin{equation}
\mathbf{p} = \left[ \frac{1.0}{1.517448}, \frac{0.367879}{1.517448}, \frac{0.149569}{1.517448} \right]^\top \approx [0.659001, 0.242433, 0.098566]^\top.
\end{equation}
