\begin{abstract}
This document formulates the mathematical properties, bilinear multiplicative interactions, and parameter parity conditions of Gated Linear Unit (GLU) variants (Standard GLU, SwiGLU, GeGLU, and ReGLU) in contemporary Transformer feedforward networks. We develop all equations across three levels of explanation: intuitive meaning, formal formulation, and practical system role. We prove the parameter budget equivalence theorem establishing the canonical $\frac{2}{3} \times 4 d_{\text{in}}$ hidden dimension adjustment, derive computational complexity bounds from primitive tensor operations, and provide an exact verified numerical trace matching the production implementation contract.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbb{R}$ denote the field of real numbers, and $\mathbb{R}^{B \times D}$ the matrix space. Matrices are denoted by uppercase bold letters ($\mathbf{X}, \mathbf{W}, \mathbf{H}, \mathbf{Y}$), vectors by lowercase bold letters ($\mathbf{x}, \mathbf{b}$), and scalar indices by standard italics. The symbol $\odot$ denotes the element-wise Hadamard product: $(\mathbf{A} \odot \mathbf{B})_{ij} = A_{ij} B_{ij}$.

\subsection{Bilinear Multiplicative Gating Operator}
\paragraph{Intuition:} Standard feedforward layers compute $\psi(\mathbf{x} \mathbf{W}_1) \mathbf{W}_2$. Gated Linear Units introduce a second parallel projection acting as an element-wise multiplicative continuous valve: one linear branch computes content, while the second branch computes a smooth gating mask that dynamically controls how much of each channel passes through.

\paragraph{Mathematical Formulation:}
For input vector $\mathbf{x} \in \mathbb{R}^{d_{\text{in}}}$, up-projection matrix $\mathbf{W}_{\text{up}} \in \mathbb{R}^{d_{\text{hidden}} \times d_{\text{in}}}$, gate matrix $\mathbf{W}_{\text{gate}} \in \mathbb{R}^{d_{\text{hidden}} \times d_{\text{in}}}$, and activation $\sigma$:
\begin{equation}
\label{eq:glu_def}
\text{GLU}(\mathbf{x}) = (\mathbf{x} \mathbf{W}_{\text{up}}^\top) \odot \sigma(\mathbf{x} \mathbf{W}_{\text{gate}}^\top) \in \mathbb{R}^{d_{\text{hidden}}}.
\end{equation}
With down-projection $\mathbf{W}_{\text{down}} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{hidden}}}$:
\begin{equation}
\label{eq:glu_ffn}
\text{FFN}_{\text{GLU}}(\mathbf{x}) = \text{GLU}(\mathbf{x}) \mathbf{W}_{\text{down}}^\top \in \mathbb{R}^{d_{\text{out}}}.
\end{equation}

\paragraph{Worked Numerical Example:}
Let $\mathbf{x} = [1.0, 2.0]$, $\mathbf{W}_{\text{up}} = \begin{bmatrix} 0.5 & 1.0 \end{bmatrix}$, $\mathbf{W}_{\text{gate}} = \begin{bmatrix} 2.0 & -1.0 \end{bmatrix}$, with $\sigma(z) = \text{ReLU}(z)$.
\begin{align}
u &= \mathbf{x} \mathbf{W}_{\text{up}}^\top = 1.0(0.5) + 2.0(1.0) = 0.5 + 2.0 = 2.5, \\
v &= \mathbf{x} \mathbf{W}_{\text{gate}}^\top = 1.0(2.0) + 2.0(-1.0) = 2.0 - 2.0 = 0.0, \\
\sigma(v) &= \text{ReLU}(0.0) = 0.0, \\
\text{ReGLU}(\mathbf{x}) &= u \odot \sigma(v) = 2.5 \times 0.0 = 0.0.
\end{align}

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size (number of tokens/samples).
    \item $d_{\text{in}} \in \mathbb{N}$: Input model dimension ($d_{\text{model}}$).
    \item $d_{\text{hidden}} \in \mathbb{N}$: Intermediate feedforward expansion width.
    \item $d_{\text{out}} \in \mathbb{N}$: Output projection dimension.
    \item $\mathbf{X} \in \mathbb{R}^{B \times d_{\text{in}}}$: Input batch matrix.
    \item $\mathbf{W}_{\text{up}} \in \mathbb{R}^{d_{\text{hidden}} \times d_{\text{in}}}$: Value/Up projection matrix ($\mathbf{W}_1$).
    \item $\mathbf{W}_{\text{gate}} \in \mathbb{R}^{d_{\text{hidden}} \times d_{\text{in}}}$: Gate projection matrix ($\mathbf{W}_2$).
    \item $\mathbf{W}_{\text{down}} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{hidden}}}$: Output projection matrix ($\mathbf{W}_3$).
\end{itemize}

\section{Problem Definition and Core Variants}
\label{sec:problem_definition}

\begin{definition}[SwiGLU Activation (Shazeer 2020)]
Using the Sigmoid Linear Unit ($\text{SiLU}(z) = z \cdot \sigma(z)$) as the gating activation:
\begin{equation}
\label{eq:swiglu}
\text{SwiGLU}(\mathbf{x}) = (\mathbf{x} \mathbf{W}_{\text{up}}^\top) \odot \text{SiLU}(\mathbf{x} \mathbf{W}_{\text{gate}}^\top).
\end{equation}
\end{definition}

\begin{definition}[GeGLU Activation]
Using the Gaussian Error Linear Unit ($\text{GELU}(z) = z \cdot \Phi(z)$) as the gating activation:
\begin{equation}
\label{eq:geglu}
\text{GeGLU}(\mathbf{x}) = (\mathbf{x} \mathbf{W}_{\text{up}}^\top) \odot \text{GELU}(\mathbf{x} \mathbf{W}_{\text{gate}}^\top).
\end{equation}
\end{definition}

\begin{definition}[ReGLU and Classical GLU]
\begin{align}
\text{ReGLU}(\mathbf{x}) &= (\mathbf{x} \mathbf{W}_{\text{up}}^\top) \odot \max(0, \mathbf{x} \mathbf{W}_{\text{gate}}^\top), \\
\text{GLU}(\mathbf{x}) &= (\mathbf{x} \mathbf{W}_{\text{up}}^\top) \odot \frac{1}{1 + e^{-\mathbf{x} \mathbf{W}_{\text{gate}}^\top}}.
\end{align}
\end{definition}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:glu} specifies the deterministic execution implemented in \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Deterministic Gated Linear Unit and Feedforward Evaluation}
\label{alg:glu}
\begin{algorithmic}[1]
\Require Input batch $\mathbf{X} \in \mathbb{R}^{B \times d_{\text{in}}}$, weights $(\mathbf{W}_{\text{gate}}, \mathbf{W}_{\text{up}}, \mathbf{W}_{\text{down}})$, variant $\in \{\text{swiglu}, \text{geglu}, \text{reglu}, \text{glu}\}$.
\Ensure Output batch $\mathbf{Y}$, gated hidden state $\mathbf{H}_{\text{gated}}$, dimensions $(B, d_{\text{in}}, d_{\text{hidden}}, d_{\text{out}})$.
\State $\mathbf{U} \gets \mathbf{X} \mathbf{W}_{\text{up}}^\top \in \mathbb{R}^{B \times d_{\text{hidden}}}$
\State $\mathbf{V} \gets \mathbf{X} \mathbf{W}_{\text{gate}}^\top \in \mathbb{R}^{B \times d_{\text{hidden}}}$
\State $\mathbf{H}_{\text{gated}} \gets \mathbf{0} \in \mathbb{R}^{B \times d_{\text{hidden}}}$
\For{$i = 1 \textbf{ to } B$}
    \For{$j = 1 \textbf{ to } d_{\text{hidden}}$}
        \State $u \gets U_{ij}, \quad v \gets V_{ij}$
        \If{$\text{variant} = \text{swiglu}$}
            \State $s \gets 1.0 / (1.0 + \exp(-v))$
            \State $H_{ij} \gets u \cdot (v \cdot s)$
        \ElsIf{$\text{variant} = \text{geglu}$}
            \State $\Phi \gets 0.5 \cdot (1.0 + \text{erf}(v / \sqrt{2}))$
            \State $H_{ij} \gets u \cdot (v \cdot \Phi)$
        \ElsIf{$\text{variant} = \text{reglu}$}
            \State $H_{ij} \gets u \cdot \max(0.0, v)$
        \ElsIf{$\text{variant} = \text{glu}$}
            \State $s \gets 1.0 / (1.0 + \exp(-v))$
            \State $H_{ij} \gets u \cdot s$
        \EndIf
    \EndFor
\EndFor
\State $\mathbf{Y} \gets \mathbf{H}_{\text{gated}} \mathbf{W}_{\text{down}}^\top \in \mathbb{R}^{B \times d_{\text{out}}}$
\State \Return $(\mathbf{Y}, \mathbf{H}_{\text{gated}})$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Parameter Parity Budgeting}
\label{sec:correctness}

\begin{theorem}[Parameter Parity under 3-Matrix Gated Architecture]
\label{thm:param_parity}
\textbf{Assumptions:}
\begin{enumerate}
    \item Standard Transformer FFN uses 2 matrices: $\mathbf{W}_1 \in \mathbb{R}^{4d \times d}$ and $\mathbf{W}_2 \in \mathbb{R}^{d \times 4d}$, with total parameters $P_{\text{std}} = 2 \times (4d \times d) = 8 d^2$.
    \item Gated FFN uses 3 matrices: $\mathbf{W}_{\text{up}}, \mathbf{W}_{\text{gate}} \in \mathbb{R}^{d_{\text{hidden}} \times d}$ and $\mathbf{W}_{\text{down}} \in \mathbb{R}^{d \times d_{\text{hidden}}}$, with total parameters $P_{\text{gated}} = 3 \times (d_{\text{hidden}} \times d) = 3 d \cdot d_{\text{hidden}}$.
\end{enumerate}

\textbf{Guarantee:}
Setting the hidden dimension to:
\begin{equation}
\label{eq:hidden_scale}
d_{\text{hidden}} = \left\lfloor \frac{2}{3} \times 4 d \right\rfloor = \left\lfloor \frac{8}{3} d \right\rfloor \approx 2.67 \, d
\end{equation}
enforces strict parameter count parity:
\begin{equation}
P_{\text{gated}} = 3 d \cdot \left( \frac{8}{3} d \right) = 8 d^2 = P_{\text{std}}.
\end{equation}

\textbf{Limitations:}
In production distributed LLMs (such as LLaMA and PaLM), $d_{\text{hidden}}$ is further rounded to the nearest multiple of 64, 128, or 256 for tensor-core alignment, resulting in a minor ($\pm 1\%$) deviation from strict equality.
\end{theorem}

\section{Complexity Derivation}
\label{sec:complexity}

\subsection{Time Complexity}
\begin{enumerate}
    \item Up-projection $\mathbf{X} \mathbf{W}_{\text{up}}^\top$: $B \cdot d_{\text{in}} \cdot d_{\text{hidden}}$ multiply-adds ($\mathcal{O}(B \cdot d_{\text{in}} \cdot d_{\text{hidden}})$).
    \item Gate-projection $\mathbf{X} \mathbf{W}_{\text{gate}}^\top$: $B \cdot d_{\text{in}} \cdot d_{\text{hidden}}$ multiply-adds ($\mathcal{O}(B \cdot d_{\text{in}} \cdot d_{\text{hidden}})$).
    \item Element-wise activation and Hadamard product: $B \cdot d_{\text{hidden}}$ scalar operations ($\mathcal{O}(B \cdot d_{\text{hidden}})$).
    \item Down-projection $\mathbf{H}_{\text{gated}} \mathbf{W}_{\text{down}}^\top$: $B \cdot d_{\text{hidden}} \cdot d_{\text{out}}$ multiply-adds ($\mathcal{O}(B \cdot d_{\text{hidden}} \cdot d_{\text{out}})$).
    \item Total time complexity:
    \begin{equation}
    T(B, d_{\text{in}}, d_{\text{hidden}}, d_{\text{out}}) = \mathcal{O}\left( B \cdot d_{\text{hidden}} \cdot (2 d_{\text{in}} + d_{\text{out}}) \right).
    \end{equation}
\end{enumerate}

\subsection{Space Complexity}
\begin{enumerate}
    \item Intermediates $\mathbf{U}, \mathbf{V}, \mathbf{H}_{\text{gated}}$ each allocate $B \cdot d_{\text{hidden}}$ scalars.
    \item Output matrix $\mathbf{Y}$ allocates $B \cdot d_{\text{out}}$ scalars.
    \item Total auxiliary storage:
    \begin{equation}
    S(B, d_{\text{hidden}}, d_{\text{out}}) = \mathcal{O}(B \cdot (d_{\text{hidden}} + d_{\text{out}})).
    \end{equation}
\end{enumerate}

\section{Worked Example and Numerical Verification}
\label{sec:worked_example}

Let $B = 1, d_{\text{in}} = 2, d_{\text{hidden}} = 2, d_{\text{out}} = 1$.
Input: $\mathbf{X} = \begin{bmatrix} 1.0 & 2.0 \end{bmatrix}$.
Weights:
\begin{equation}
\mathbf{W}_{\text{up}} = \begin{bmatrix} 0.5 & -1.0 \\ 1.0 & 0.0 \end{bmatrix}, \quad
\mathbf{W}_{\text{gate}} = \begin{bmatrix} 1.0 & 0.0 \\ 0.0 & 1.0 \end{bmatrix}, \quad
\mathbf{W}_{\text{down}} = \begin{bmatrix} 2.0 & 0.5 \end{bmatrix}.
\end{equation}
Variant: \texttt{reglu} ($\sigma(z) = \max(0, z)$).

\subsection{Execution Trace}
\begin{align}
\mathbf{U} = \mathbf{X} \mathbf{W}_{\text{up}}^\top &= [1.0(0.5) + 2.0(-1.0), \; 1.0(1.0) + 2.0(0.0)] = [0.5 - 2.0, \; 1.0 + 0.0] = [-1.5, \; 1.0]. \\
\mathbf{V} = \mathbf{X} \mathbf{W}_{\text{gate}}^\top &= [1.0(1.0) + 2.0(0.0), \; 1.0(0.0) + 2.0(1.0)] = [1.0, \; 2.0]. \\
\sigma(\mathbf{V}) = \text{ReLU}(\mathbf{V}) &= [\max(0, 1.0), \; \max(0, 2.0)] = [1.0, \; 2.0]. \\
\mathbf{H}_{\text{gated}} = \mathbf{U} \odot \sigma(\mathbf{V}) &= [-1.5 \times 1.0, \; 1.0 \times 2.0] = [-1.5, \; 2.0]. \\
\mathbf{Y} = \mathbf{H}_{\text{gated}} \mathbf{W}_{\text{down}}^\top &= [-1.5(2.0) + 2.0(0.5)] = [-3.0 + 1.0] = [-2.0].
\end{align}
