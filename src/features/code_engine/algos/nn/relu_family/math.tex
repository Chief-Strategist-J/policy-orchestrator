\begin{abstract}
This document establishes the mathematical foundations, subdifferential formulations, and stability characteristics of the Rectified Linear Unit (ReLU) family of activation functions. We formulate the piecewise linear mappings (Standard ReLU, Leaky ReLU, Parametric ReLU) and exponential smooth variations (ELU, Self-Normalizing SELU). Every formula is explained at three levels: intuitive meaning, formal formulation, and practical application. We analyze the ``dying ReLU'' failure mode, state the SELU Fixed-Point Normalization Theorem with explicit assumptions, guarantees, and limitations, and provide an exact numerical trace matching the production implementation contract.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbb{R}$ denote the set of real numbers, $\mathbb{R}^D$ the $D$-dimensional Euclidean space, and $\mathbb{R}^{N \times D}$ the space of real matrices of shape $(N, D)$. Scalars are denoted in lowercase standard type ($x, z, \alpha, \lambda$), vectors in lowercase bold ($\mathbf{x}, \mathbf{z}, \mathbf{a}$), and matrices in uppercase bold ($\mathbf{X}, \mathbf{Y}$).

\subsection{Piecewise and Non-Linear Activation Operators}
\paragraph{Intuition:} Activation functions apply a non-linear threshold or curvature to each scalar coordinate of a pre-activation tensor. The ReLU family operates by passing positive inputs linearly while attenuating, zeroing, or softly compressing negative inputs to enable non-linear representation without saturating on positive signals.

\paragraph{Mathematical Formulation:}
For a tensor $\mathbf{X} \in \mathbb{R}^{N \times D}$, an element-wise activation $\psi: \mathbb{R} \to \mathbb{R}$ produces transformed tensor $\mathbf{Y} \in \mathbb{R}^{N \times D}$ where:
\begin{equation}
Y_{ij} = \psi(X_{ij}), \quad \forall i \in \{1, \dots, N\}, \; j \in \{1, \dots, D\}.
\end{equation}

\paragraph{Worked Numerical Example:}
Let $x_1 = 3.0$ and $x_2 = -2.0$:
\begin{align}
\text{ReLU}(3.0) &= \max(0, 3.0) = 3.0, \quad \text{ReLU}(-2.0) = \max(0, -2.0) = 0.0, \\
\text{LeakyReLU}(3.0; \alpha=0.01) &= 3.0, \quad \text{LeakyReLU}(-2.0; \alpha=0.01) = 0.01(-2.0) = -0.02.
\end{align}

\subsection{Summary of Symbols}
\begin{itemize}
    \item $N \in \mathbb{N}$: Batch dimension (number of independent samples).
    \item $D \in \mathbb{N}$: Feature / channel dimension.
    \item $\mathbf{X} \in \mathbb{R}^{N \times D}$: Input pre-activation tensor.
    \item $\mathbf{Y} \in \mathbb{R}^{N \times D}$: Output activated tensor.
    \item $\alpha \in \mathbb{R}_{>0}$: Slope or scaling coefficient for negative inputs.
    \item $\mathbf{a} \in \mathbb{R}^D$: Channel-wise learned slope parameter vector in PReLU.
    \item $\lambda_{\text{SELU}} \approx 1.050700987$: Fixed SELU scale constant.
    \item $\alpha_{\text{SELU}} \approx 1.673263242$: Fixed SELU exponential saturation constant.
    \item $\partial \psi(x)$: Clarke subdifferential / generalized gradient at point $x$.
\end{itemize}

\section{Problem Definition and Core Variants}
\label{sec:problem_definition}

\begin{definition}[Standard Rectified Linear Unit (ReLU)]
The standard ReLU function is defined as:
\begin{equation}
\label{eq:relu}
\text{ReLU}(x) = \max(0, x) = \begin{cases} x & \text{if } x > 0, \\ 0 & \text{if } x \le 0. \end{cases}
\end{equation}
Its subgradient is:
\begin{equation}
\partial \text{ReLU}(x) = \begin{cases} \{1\} & \text{if } x > 0, \\ [0, 1] & \text{if } x = 0, \\ \{0\} & \text{if } x < 0. \end{cases}
\end{equation}
\end{definition}

\begin{definition}[Leaky ReLU]
For a fixed slope parameter $\alpha > 0$ (typically $\alpha = 0.01$):
\begin{equation}
\label{eq:leaky_relu}
\text{LeakyReLU}(x; \alpha) = \max(\alpha x, x) = \begin{cases} x & \text{if } x > 0, \\ \alpha x & \text{if } x \le 0. \end{cases}
\end{equation}
\end{definition}

\begin{definition}[Parametric ReLU (PReLU)]
For a channel coordinate $j \in \{1, \dots, D\}$ with learned parameter $a_j$:
\begin{equation}
\label{eq:prelu}
\text{PReLU}(x_j; a_j) = \begin{cases} x_j & \text{if } x_j > 0, \\ a_j x_j & \text{if } x_j \le 0. \end{cases}
\end{equation}
\end{definition}

\begin{definition}[Exponential Linear Unit (ELU)]
For scale parameter $\alpha > 0$ (typically $\alpha = 1.0$):
\begin{equation}
\label{eq:elu}
\text{ELU}(x; \alpha) = \begin{cases} x & \text{if } x > 0, \\ \alpha (e^x - 1) & \text{if } x \le 0. \end{cases}
\end{equation}
ELU is continuously differentiable everywhere when $\alpha = 1.0$, since $\lim_{x \to 0^-} \frac{d}{dx}[\alpha (e^x - 1)] = \alpha = 1 = \lim_{x \to 0^+} \frac{d}{dx}[x]$.
\end{definition}

\begin{definition}[Scaled Exponential Linear Unit (SELU)]
Using specific mathematical constants $\lambda \approx 1.0507$ and $\alpha \approx 1.6733$:
\begin{equation}
\label{eq:selu}
\text{SELU}(x) = \lambda \begin{cases} x & \text{if } x > 0, \\ \alpha (e^x - 1) & \text{if } x \le 0. \end{cases}
\end{equation}
\end{definition}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:relu_family} defines the complete deterministic element-wise execution across all five variants.

\begin{algorithm}[H]
\caption{Deterministic ReLU Family Tensor Activation}
\label{alg:relu_family}
\begin{algorithmic}[1]
\Require Input tensor $\mathbf{X} \in \mathbb{R}^{N \times D}$, variant identifier $\in \{\text{relu}, \text{leaky\_relu}, \text{prelu}, \text{elu}, \text{selu}\}$, parameter $\alpha$, channel weights $\mathbf{a} \in \mathbb{R}^D$.
\Ensure Output tensor $\mathbf{Y} \in \mathbb{R}^{N \times D}$, dead neuron fraction $\rho_{\text{dead}}$, mean activation $\mu$, variance $\sigma^2$.
\State $N \gets \text{rows}(\mathbf{X}), \quad D \gets \text{cols}(\mathbf{X})$
\State $\mathbf{Y} \gets \mathbf{0} \in \mathbb{R}^{N \times D}, \quad \text{zero\_count} \gets 0$
\For{$i = 1 \textbf{ to } N$}
    \For{$j = 1 \textbf{ to } D$}
        \State $x \gets X_{ij}$
        \If{$\text{variant} = \text{relu}$}
            \State $y \gets \max(0.0, x)$
            \If{$y = 0.0$} $\text{zero\_count} \gets \text{zero\_count} + 1$ \EndIf
        \ElsIf{$\text{variant} = \text{leaky\_relu}$}
            \State $y \gets x \textbf{ if } x > 0.0 \textbf{ else } \alpha \cdot x$
        \ElsIf{$\text{variant} = \text{prelu}$}
            \State $y \gets x \textbf{ if } x > 0.0 \textbf{ else } a_j \cdot x$
        \ElsIf{$\text{variant} = \text{elu}$}
            \State $y \gets x \textbf{ if } x > 0.0 \textbf{ else } \alpha \cdot (e^x - 1.0)$
        \ElsIf{$\text{variant} = \text{selu}$}
            \State $\lambda \gets 1.0507009873554805, \quad \alpha_s \gets 1.6732632423543772$
            \State $y \gets \lambda \cdot (x \textbf{ if } x > 0.0 \textbf{ else } \alpha_s \cdot (e^x - 1.0))$
        \EndIf
        \State $Y_{ij} \gets y$
    \EndFor
\EndFor
\State $\rho_{\text{dead}} \gets \frac{\text{zero\_count}}{N \cdot D}$
\State Compute tensor mean $\mu = \frac{1}{ND}\sum Y_{ij}$ and variance $\sigma^2 = \frac{1}{ND}\sum (Y_{ij} - \mu)^2$
\State \Return $(\mathbf{Y}, \rho_{\text{dead}}, \mu, \sigma^2)$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Dying ReLU and Self-Normalization}
\label{sec:correctness}

\subsection{The Dying ReLU Phenomenon}
\begin{property}[Zero Gradient Trap]
For standard ReLU, if a neuron's weights evolve such that for all training samples $\mathbf{x} \in \mathcal{S}$, the pre-activation $z = \mathbf{w}^\top \mathbf{x} + b \le 0$, then $\partial \text{ReLU}(z) = 0$. In gradient descent:
\begin{equation}
\nabla_{\mathbf{w}} \mathcal{L} = \frac{\partial \mathcal{L}}{\partial y} \frac{\partial y}{\partial z} \mathbf{x} = \frac{\partial \mathcal{L}}{\partial y} \cdot 0 \cdot \mathbf{x} = \mathbf{0}.
\end{equation}
The neuron receives zero gradient and will never update or recover (permanently dead unit).
\end{property}

\subsection{SELU Fixed-Point Normalization Theorem}
\begin{theorem}[Klambauer et al. 2017: Self-Normalizing Neural Networks]
\label{thm:selu}
\textbf{Assumptions:}
\begin{enumerate}
    \item Input activations follow a distribution with mean $\mu = 0$ and variance $\sigma^2 = 1$.
    \item Weights are initialized with zero mean and variance $\text{Var}(W) = 1/d_{\text{in}}$ (LeCun initialization).
    \item Activation is SELU with fixed constants $\lambda \approx 1.0507$ and $\alpha \approx 1.6733$.
\end{enumerate}

\textbf{Guarantee:}
The transformation mapping layer mean and variance $(\mu, \sigma^2) \to (\tilde{\mu}, \tilde{\sigma}^2)$ has a stable fixed point at $(\mu^* = 0, (\sigma^2)^* = 1)$. Activations and gradients remain self-normalized across hundreds of layers without explicit batch normalization.

\textbf{Limitations:}
\begin{enumerate}
    \item The fixed point is strictly stable only under LeCun variance scaling; using standard Glorot or He scaling breaks the self-normalizing property.
    \item Dropout or standard batch normalization disrupt the theoretical fixed-point dynamics.
\end{enumerate}
\end{theorem}

\section{Complexity Derivation}
\label{sec:complexity}

\subsection{Time Complexity}
\begin{enumerate}
    \item For a tensor of shape $(N, D)$, the nested loops execute exactly $N \cdot D$ times.
    \item For ReLU, Leaky ReLU, and PReLU, each element requires $\mathcal{O}(1)$ basic scalar comparison and multiplication.
    \item For ELU and SELU, each negative element requires $\mathcal{O}(1)$ transcendental exponential evaluation ($e^x$).
    \item Total time complexity is strictly linear in the total number of tensor elements:
    \begin{equation}
    T(N, D) = \mathcal{O}(N \cdot D).
    \end{equation}
\end{enumerate}

\subsection{Space Complexity}
\begin{enumerate}
    \item Output tensor $\mathbf{Y}$ allocates $N \cdot D$ floating-point scalars.
    \item Space complexity is strictly:
    \begin{equation}
    S(N, D) = \mathcal{O}(N \cdot D).
    \end{equation}
\end{enumerate}

\section{Worked Example and Numerical Trace}
\label{sec:worked_example}

Consider an input tensor $\mathbf{X} \in \mathbb{R}^{2 \times 3}$:
\begin{equation}
\mathbf{X} = \begin{bmatrix} 2.0 & -1.0 & 0.0 \\ -3.0 & 1.5 & -0.5 \end{bmatrix}.
\end{equation}

\subsection{Element-Wise Calculations Across Variants}
\paragraph{1. Standard ReLU ($\mathbf{Y} = \max(0, \mathbf{X})$):}
\begin{equation}
\mathbf{Y}_{\text{relu}} = \begin{bmatrix} \max(0, 2.0) & \max(0, -1.0) & \max(0, 0.0) \\ \max(0, -3.0) & \max(0, 1.5) & \max(0, -0.5) \end{bmatrix} = \begin{bmatrix} 2.0 & 0.0 & 0.0 \\ 0.0 & 1.5 & 0.0 \end{bmatrix}.
\end{equation}
Dead neuron count: $4$ out of $6 \implies \rho_{\text{dead}} = 4/6 \approx 0.6667$.

\paragraph{2. Leaky ReLU ($\alpha = 0.01$):}
\begin{equation}
\mathbf{Y}_{\text{leaky}} = \begin{bmatrix} 2.0 & 0.01(-1.0) & 0.01(0.0) \\ 0.01(-3.0) & 1.5 & 0.01(-0.5) \end{bmatrix} = \begin{bmatrix} 2.0 & -0.01 & 0.0 \\ -0.03 & 1.5 & -0.005 \end{bmatrix}.
\end{equation}

\paragraph{3. ELU ($\alpha = 1.0$):}
For negative entries: $e^{-1.0} - 1 \approx 0.367879 - 1 = -0.632121$, $e^{-3.0} - 1 \approx 0.049787 - 1 = -0.950213$, $e^{-0.5} - 1 \approx 0.606531 - 1 = -0.393469$.
\begin{equation}
\mathbf{Y}_{\text{elu}} \approx \begin{bmatrix} 2.0 & -0.6321 & 0.0 \\ -0.9502 & 1.5 & -0.3935 \end{bmatrix}.
\end{equation}

\section{Numerical Considerations and Stability}
\label{sec:numerical}

\begin{enumerate}
    \item \textbf{Underflow in Exponentials:} For ELU and SELU, when $x < -50.0$, $e^x \to 0$ in IEEE 754 float64 arithmetic, correctly saturating to $-\alpha$ or $-\lambda \alpha$ without underflow exceptions.
    \item \textbf{Channel Dimension Matching for PReLU:} The implementation enforces $\text{len}(\mathbf{a}) = D$, ensuring 1-to-1 channel slope assignment.
\end{enumerate}
