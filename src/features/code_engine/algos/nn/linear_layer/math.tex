\begin{abstract}
This document presents the mathematical foundation, matrix algebraic formulations, and statistical variance preservation theory for linear (affine) projection layers in deep neural architectures. Every mathematical operation is presented at three levels of explanation: intuition, formal formulation, and practical application. We provide a step-by-step derivation of forward and backward signal variance scaling, state the Glorot (Xavier), He (Kaiming), and LeCun theorems with explicitly separated assumptions, guarantees, and limitations, derive computational complexity bounds from primitive operations, and provide a verified numerical trace matching the production implementation contract.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbb{R}$ denote the set of real numbers, and $\mathbb{N} = \{1, 2, \dots\}$. Matrices are denoted by uppercase bold letters ($\mathbf{X}, \mathbf{W}, \mathbf{Y}$), vectors by lowercase bold letters ($\mathbf{x}, \mathbf{b}, \mathbf{y}$), and scalar indices by standard italics.

\subsection{Matrix Multiplication with Transposed Weights}
\paragraph{Intuition:} In a batched feedforward layer, each row of the input batch represents an independent data sample, and each row of the weight matrix represents the linear weights for one output feature channel. Multiplying the batch by the transposed weight matrix projects all samples simultaneously into the output channel space.

\paragraph{Mathematical Formulation:}
Let $\mathbf{X} \in \mathbb{R}^{B \times d_{\text{in}}}$ be a batch of $B$ samples, $\mathbf{W} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$ be the weight matrix, and $\mathbf{b} \in \mathbb{R}^{d_{\text{out}}}$ be the bias vector. The batched affine transformation is defined as:
\begin{equation}
\label{eq:affine_mat}
\mathbf{Y} = \mathbf{X} \mathbf{W}^\top + \mathbf{1}_B \mathbf{b}^\top \in \mathbb{R}^{B \times d_{\text{out}}},
\end{equation}
where $\mathbf{1}_B = [1, 1, \dots, 1]^\top \in \mathbb{R}^B$. For each entry $(i, j)$ with $i \in \{1, \dots, B\}$ and $j \in \{1, \dots, d_{\text{out}}\}$:
\begin{equation}
\label{eq:scalar_expansion}
Y_{ij} = \sum_{k=1}^{d_{\text{in}}} X_{ik} W_{jk} + b_j.
\end{equation}

\paragraph{Worked Numerical Example:}
Let $B = 2, d_{\text{in}} = 3, d_{\text{out}} = 2$:
\begin{equation}
\mathbf{X} = \begin{bmatrix} 1.0 & 2.0 & -1.0 \\ 0.0 & -0.5 & 1.5 \end{bmatrix}, \quad
\mathbf{W} = \begin{bmatrix} 0.5 & -1.0 & 2.0 \\ -2.0 & 0.0 & 1.0 \end{bmatrix}, \quad
\mathbf{b} = \begin{bmatrix} 0.25 \\ -0.50 \end{bmatrix}.
\end{equation}
For sample 1 ($i=1$):
\begin{align}
Y_{11} &= X_{11} W_{11} + X_{12} W_{12} + X_{13} W_{13} + b_1 = 1.0(0.5) + 2.0(-1.0) + (-1.0)(2.0) + 0.25 = 0.5 - 2.0 - 2.0 + 0.25 = -3.25, \\
Y_{12} &= X_{11} W_{21} + X_{12} W_{22} + X_{13} W_{23} + b_2 = 1.0(-2.0) + 2.0(0.0) + (-1.0)(1.0) - 0.50 = -2.0 + 0.0 - 1.0 - 0.50 = -3.50.
\end{align}
For sample 2 ($i=2$):
\begin{align}
Y_{21} &= 0.0(0.5) + (-0.5)(-1.0) + 1.5(2.0) + 0.25 = 0.0 + 0.5 + 3.0 + 0.25 = 3.75, \\
Y_{22} &= 0.0(-2.0) + (-0.5)(0.0) + 1.5(1.0) - 0.50 = 0.0 + 0.0 + 1.5 - 0.50 = 1.00.
\end{align}
Output matrix:
\begin{equation}
\mathbf{Y} = \begin{bmatrix} -3.25 & -3.50 \\ 3.75 & 1.00 \end{bmatrix}.
\end{equation}

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size (number of independent examples).
    \item $d_{\text{in}} \in \mathbb{N}$: Fan-in dimension (number of input features / columns of $\mathbf{X}$ and $\mathbf{W}$).
    \item $d_{\text{out}} \in \mathbb{N}$: Fan-out dimension (number of output neurons / rows of $\mathbf{W}$).
    \item $\mathbf{X} \in \mathbb{R}^{B \times d_{\text{in}}}$: Input batch matrix.
    \item $\mathbf{W} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$: Trainable weight matrix.
    \item $\mathbf{b} \in \mathbb{R}^{d_{\text{out}}}$: Trainable bias vector.
    \item $\mathbf{Y} \in \mathbb{R}^{B \times d_{\text{out}}}$: Output activation matrix.
    \item $\sigma^2 \in \mathbb{R}_{>0}$: Theoretical weight initialization variance.
    \item $r \in \mathbb{R}_{>0}$: Uniform initialization half-width boundary ($W_{jk} \sim \mathcal{U}[-r, r]$).
\end{itemize}

\section{Problem Definition and Core Concepts}
\label{sec:problem_definition}

\subsection{Fan-In and Fan-Out Metrics}
\begin{definition}[Fan-in and Fan-out]
For a linear transformation with weight matrix $\mathbf{W} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$:
\begin{align}
\text{fan\_in} &= d_{\text{in}} = \text{number of incoming connections per neuron}, \\
\text{fan\_out} &= d_{\text{out}} = \text{number of outgoing connections per input channel}.
\end{align}
\end{definition}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:linear_layer} defines the deterministic evaluation and initialization metric derivation.

\begin{algorithm}[H]
\caption{Affine Matrix Multiplication and Variance Scaling Calculation}
\label{alg:linear_layer}
\begin{algorithmic}[1]
\Require Input batch $\mathbf{X} \in \mathbb{R}^{B \times d_{\text{in}}}$, weight matrix $\mathbf{W} \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$, bias $\mathbf{b} \in \mathbb{R}^{d_{\text{out}}}$, mode $\in \{\text{kaiming\_uniform}, \text{glorot\_uniform}, \text{lecun\_uniform}\}$.
\Ensure Transformed matrix $\mathbf{Y} \in \mathbb{R}^{B \times d_{\text{out}}}$, dimensions $(B, d_{\text{in}}, d_{\text{out}})$, variance $\sigma^2$, bound $r$.
\State $B \gets \text{rows}(\mathbf{X}), \quad d_{\text{in}} \gets \text{cols}(\mathbf{X}), \quad d_{\text{out}} \gets \text{rows}(\mathbf{W})$
\State $\mathbf{Y} \gets \mathbf{0} \in \mathbb{R}^{B \times d_{\text{out}}}$
\For{$i = 1 \textbf{ to } B$}
    \For{$j = 1 \textbf{ to } d_{\text{out}}$}
        \State $\text{sum} \gets b_j$
        \For{$k = 1 \textbf{ to } d_{\text{in}}$}
            \State $\text{sum} \gets \text{sum} + X_{ik} \cdot W_{jk}$
        \EndFor
        \State $Y_{ij} \gets \text{sum}$
    \EndFor
\EndFor
\If{$\text{mode} = \text{glorot\_uniform}$}
    \State $\sigma^2 \gets \frac{2}{d_{\text{in}} + d_{\text{out}}}, \quad r \gets \sqrt{3 \sigma^2} = \sqrt{\frac{6}{d_{\text{in}} + d_{\text{out}}}}$
\ElsIf{$\text{mode} = \text{kaiming\_uniform}$}
    \State $\sigma^2 \gets \frac{2}{d_{\text{in}}}, \quad r \gets \sqrt{3 \sigma^2} = \sqrt{\frac{6}{d_{\text{in}}}}$
\ElsIf{$\text{mode} = \text{lecun\_uniform}$}
    \State $\sigma^2 \gets \frac{1}{d_{\text{in}}}, \quad r \gets \sqrt{3 \sigma^2} = \sqrt{\frac{3}{d_{\text{in}}}}$
\EndIf
\State \Return $(\mathbf{Y}, B, d_{\text{in}}, d_{\text{out}}, \sigma^2, r)$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Variance Scaling and Initialization}
\label{sec:correctness}

\subsection{Derivation of Signal Variance Preservation}
Consider a linear layer $y = \sum_{k=1}^{d_{\text{in}}} w_k x_k$. Assume:
\begin{enumerate}
    \item Input elements $x_k$ are mutually independent, identically distributed with $\mathbb{E}[x_k] = 0$ and $\text{Var}(x_k) = \text{Var}(x)$.
    \item Weight elements $w_k$ are mutually independent, identically distributed with $\mathbb{E}[w_k] = 0$ and $\text{Var}(w_k) = \text{Var}(w)$.
    \item Inputs and weights are mutually independent.
\end{enumerate}

\paragraph{Step-by-Step Variance Derivation:}
\begin{equation}
\mathbb{E}[y] = \sum_{k=1}^{d_{\text{in}}} \mathbb{E}[w_k x_k] = \sum_{k=1}^{d_{\text{in}}} \mathbb{E}[w_k] \mathbb{E}[x_k] = 0.
\end{equation}
For the variance of the sum of independent random variables:
\begin{align}
\text{Var}(y) &= \sum_{k=1}^{d_{\text{in}}} \text{Var}(w_k x_k) \nonumber \\
&= \sum_{k=1}^{d_{\text{in}}} \left( \mathbb{E}[(w_k x_k)^2] - (\mathbb{E}[w_k x_k])^2 \right) \nonumber \\
&= \sum_{k=1}^{d_{\text{in}}} \mathbb{E}[w_k^2] \mathbb{E}[x_k^2] \nonumber \\
&= d_{\text{in}} \text{Var}(w) \text{Var}(x). \label{eq:var_forward}
\end{align}
To preserve signal scale across deep architectures ($\text{Var}(y) = \text{Var}(x)$), we require:
\begin{equation}
\text{Var}(w) = \frac{1}{d_{\text{in}}}.
\end{equation}

\begin{theorem}[Initialization Schemes Comparison]
\label{thm:init_schemes}
\textbf{Assumptions:}
Linear layer operations with zero-mean inputs and weights, independent coordinates, and standard activation assumptions:
\begin{enumerate}
    \item \textbf{LeCun Scheme (Linear / Tanh):} Assumes linear activations ($\sigma(z) = z$). Enforces forward variance preservation: $\text{Var}(w) = \frac{1}{d_{\text{in}}}$.
    \item \textbf{Glorot / Xavier Scheme (Symmetric Activations):} Balances forward signal propagation ($\text{Var}(w) = 1/d_{\text{in}}$) and backward gradient propagation ($\text{Var}(w) = 1/d_{\text{out}}$) by taking the harmonic mean:
    \begin{equation}
    \sigma^2 = \frac{2}{d_{\text{in}} + d_{\text{out}}}.
    \end{equation}
    \item \textbf{He / Kaiming Scheme (Rectified Activations):} Assumes ReLU activation ($\sigma(z) = \max(0, z)$), where exactly half of the activations are zeroed out in expectation ($\mathbb{E}[\text{ReLU}(z)^2] = \frac{1}{2} \text{Var}(z)$). To compensate for the $1/2$ variance loss:
    \begin{equation}
    \sigma^2 = \frac{2}{d_{\text{in}}}.
    \end{equation}
\end{enumerate}

\textbf{Guarantee:}
Drawing weights from distribution with variance $\sigma^2$ guarantees that forward activation variances and backward gradient variances neither explode exponentially nor vanish to zero at layer $0$.

\textbf{Limitations:}
\begin{enumerate}
    \item Assumptions of independence and zero-mean distributions break down under correlated features and batch normalization shifts.
    \item Residual connections introduce additive variance growth ($x + F(x)$), requiring additional branch scale adjustments ($1/\sqrt{2L}$).
\end{enumerate}
\end{theorem}

\section{Complexity Derivation}
\label{sec:complexity}

\subsection{Time Complexity}
\begin{enumerate}
    \item Evaluating equation~\eqref{eq:scalar_expansion} for $B$ rows and $d_{\text{out}}$ columns:
    \item For each $(i, j)$, the inner loop computes $d_{\text{in}}$ multiplications and $d_{\text{in}}$ additions ($2 d_{\text{in}}$ operations).
    \item Adding bias $b_j$ takes $1$ addition.
    \item Total elementary operations: $B \cdot d_{\text{out}} \cdot (2 d_{\text{in}} + 1) = \Theta(B \cdot d_{\text{in}} \cdot d_{\text{out}})$.
    \begin{equation}
    T(B, d_{\text{in}}, d_{\text{out}}) = \mathcal{O}(B \cdot d_{\text{in}} \cdot d_{\text{out}}).
    \end{equation}
\end{enumerate}

\subsection{Space Complexity}
\begin{enumerate}
    \item Output matrix $\mathbf{Y}$ allocates $B \cdot d_{\text{out}}$ floating-point numbers.
    \item Auxiliary memory complexity is strictly:
    \begin{equation}
    S(B, d_{\text{out}}) = \mathcal{O}(B \cdot d_{\text{out}}).
    \end{equation}
\end{enumerate}

\section{Numerical Considerations and Boundary Behaviors}
\label{sec:numerical}

\begin{enumerate}
    \item \textbf{Row-Major Memory Locality:} Matrix multiplications accumulate inner products along contiguous column strides ($k = 1 \dots d_{\text{in}}$), maximizing cache hit rates.
    \item \textbf{Null Bias Handling:} When bias vector $\mathbf{b}$ is omitted or \texttt{null}, bias addition defaults to zero with zero overhead.
    \item \textbf{Dimensionality Constraints:} Validates that $\text{cols}(\mathbf{X}) = d_{\text{in}} = \text{cols}(\mathbf{W})$ and $\text{len}(\mathbf{b}) = d_{\text{out}}$.
\end{enumerate}
