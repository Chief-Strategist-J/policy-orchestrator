\begin{abstract}
This document establishes the algebraic foundations, index-gathering operators, sparse gradient mechanics, and sequence pooling operations for neural embedding lookup tables. We prove the exact equivalence between discrete indexing gather operations and one-hot matrix multiplication, derive asymptotic memory and computational advantages, formulate mean/sum/sqrt-N aggregation functions, and provide an exact worked trace matching the production implementation contract.
\end{abstract}

\section{Notation}
\label{sec:notation}

Let $\mathbb{R}$ denote the set of real numbers, $\mathbb{N}_0 = \{0, 1, 2, \dots\}$, and $\mathbb{N} = \{1, 2, \dots\}$. Matrices are denoted by uppercase bold letters ($\mathbf{E}, \mathbf{Y}$), vectors by lowercase bold letters ($\mathbf{e}_i, \mathbf{v}$), and index sets by bold indices ($\mathbf{I}$).

\begin{itemize}
    \item $V \in \mathbb{N}$: Vocabulary size (number of discrete entities / rows).
    \item $D \in \mathbb{N}$: Embedding feature dimension (width of each continuous vector).
    \item $B \in \mathbb{N}$: Batch size.
    \item $S \in \mathbb{N}$: Sequence length (number of categorical tokens per batch item).
    \item $\mathbf{E} \in \mathbb{R}^{V \times D}$: Master embedding lookup table.
    \item $\mathbf{I} \in \{0, \dots, V-1\}^{B \times S}$: 2D integer token index matrix.
    \item $\mathbf{Y} \in \mathbb{R}^{B \times S \times D}$: Output 3D gathered embedding tensor.
    \item $\mathbf{1}_{k} \in \{0, 1\}^V$: Canonical standard basis indicator vector (one-hot vector with 1 at coordinate $k$).
    \item $\text{pad} \in \{0, \dots, V-1\} \cup \{\text{null}\}$: Optional padding index representing masked tokens.
\end{itemize}

\section{Problem Definition}
\label{sec:problem_definition}

\begin{definition}[Discrete Gathering Embedding Lookup Operator]
Given vocabulary matrix $\mathbf{E} = [\mathbf{v}_0, \mathbf{v}_1, \dots, \mathbf{v}_{V-1}]^\top \in \mathbb{R}^{V \times D}$ where $\mathbf{v}_k \in \mathbb{R}^D$, and index matrix $\mathbf{I} \in \{0, \dots, V-1\}^{B \times S}$, the embedding lookup operator $\mathcal{L}_{\mathbf{E}}: \{0, \dots, V-1\}^{B \times S} \to \mathbb{R}^{B \times S \times D}$ maps discrete coordinates to continuous representations:
\begin{equation}
\label{eq:embedding_lookup}
Y_{b, s, :} = \mathcal{L}_{\mathbf{E}}(\mathbf{I})_{b, s, :} = \begin{cases} \mathbf{0}_D & \text{if } I_{b, s} = \text{pad}, \\ \mathbf{v}_{I_{b, s}} = \mathbf{E}_{I_{b, s}, :} & \text{otherwise}, \end{cases} \quad \forall b \in \{1, \dots, B\}, \; s \in \{1, \dots, S\}.
\end{equation}
\end{definition}

\begin{definition}[Bag-of-Embeddings Sequence Aggregators]
For sentence or entity pooling across sequence length $S$, the combiner maps sequence $\mathbf{Y}_b \in \mathbb{R}^{S \times D}$ to pooled representation $\bar{\mathbf{y}}_b \in \mathbb{R}^D$:
\begin{align}
\bar{\mathbf{y}}_b^{\text{sum}} &= \sum_{s=1}^S Y_{b, s, :}, \label{eq:pool_sum} \\
\bar{\mathbf{y}}_b^{\text{mean}} &= \frac{1}{|S_{\text{valid}}|} \sum_{s \in S_{\text{valid}}} Y_{b, s, :}, \label{eq:pool_mean} \\
\bar{\mathbf{y}}_b^{\text{sqrtn}} &= \frac{1}{\sqrt{|S_{\text{valid}}|}} \sum_{s \in S_{\text{valid}}} Y_{b, s, :}. \label{eq:pool_sqrtn}
\end{align}
\end{definition}

\section{Algorithm}
\label{sec:algorithm}

Algorithm~\ref{alg:embedding_lookup} details the deterministic gather and pooling execution.

\begin{algorithm}[H]
\caption{Embedding Lookup Gathering and Bag Aggregation}
\label{alg:embedding_lookup}
\begin{algorithmic}[1]
\Require Index matrix $\mathbf{I} \in \mathbb{N}_0^{B \times S}$, table $\mathbf{E} \in \mathbb{R}^{V \times D}$, padding index $\text{pad}$, combiner $\in \{\text{none}, \text{mean}, \text{sum}, \text{sqrtn}\}$.
\Ensure 3D embedding tensor $\mathbf{Y} \in \mathbb{R}^{B \times S \times D}$, optional pooled matrix $\bar{\mathbf{Y}} \in \mathbb{R}^{B \times D}$, sparse gradient mask $\mathbf{m} \in \{0, 1\}^V$.
\State Initialize $\mathbf{Y} \in \mathbb{R}^{B \times S \times D}$ and mask $\mathbf{m} \gets \mathbf{0}_V$
\For{$b = 1 \textbf{ to } B$}
    \For{$s = 1 \textbf{ to } S$}
        \State $k \gets I_{bs}$
        \State $m_k \gets 1$
        \If{$k = \text{pad}$}
            \State $\mathbf{Y}_{b, s, :} \gets \mathbf{0}_D$
        \Else
            \State $\mathbf{Y}_{b, s, :} \gets \mathbf{E}_{k, :}$
        \EndIf
    \EndFor
\EndFor
\If{$\text{combiner} \ne \text{none}$}
    \State Compute pooled batch $\bar{\mathbf{Y}} \in \mathbb{R}^{B \times D}$ via Equation~\eqref{eq:pool_sum}, \eqref{eq:pool_mean}, or \eqref{eq:pool_sqrtn}
\EndIf
\State \Return $(\mathbf{Y}, \bar{\mathbf{Y}}, \mathbf{m})$
\end{algorithmic}
\end{algorithm}

\section{Correctness and Sparse Gradient Equivalence}
\label{sec:correctness}

\begin{theorem}[One-Hot Matrix Multiplication Equivalence]
\label{thm:one_hot_equiv}
Let $\mathbf{1}_{k} \in \{0, 1\}^V$ be the one-hot representation of index $k \in \{0, \dots, V-1\}$ where coordinate $j = k$ is 1 and all other coordinates are 0. Then the linear product $\mathbf{1}_k^\top \mathbf{E}$ is identically equal to the gathered vector $\mathbf{E}_{k, :}$:
\begin{equation}
\mathbf{1}_k^\top \mathbf{E} = \sum_{j=0}^{V-1} (\mathbf{1}_k)_j \mathbf{E}_{j, :} = 1 \cdot \mathbf{E}_{k, :} + \sum_{j \ne k} 0 \cdot \mathbf{E}_{j, :} = \mathbf{E}_{k, :}.
\end{equation}
\end{theorem}

\begin{theorem}[Sparse Gradient Update Bound]
\label{thm:sparse_gradient}
Let loss $\mathcal{L}$ receive downstream gradients $\frac{\partial \mathcal{L}}{\partial \mathbf{Y}} \in \mathbb{R}^{B \times S \times D}$. The full gradient with respect to embedding table $\mathbf{E} \in \mathbb{R}^{V \times D}$ is sparse:
\begin{equation}
\frac{\partial \mathcal{L}}{\partial \mathbf{E}_{k, :}} = \sum_{b=1}^B \sum_{s=1}^S \mathbf{1}_{\{I_{bs} = k\}} \frac{\partial \mathcal{L}}{\partial Y_{b, s, :}}.
\end{equation}
The number of non-zero rows in $\frac{\partial \mathcal{L}}{\partial \mathbf{E}}$ is bounded by $\min(V, B \cdot S) \ll V$, enabling $\mathcal{O}(B \cdot S \cdot D)$ sparse optimizer updates.
\end{theorem}

\section{Complexity Analysis}
\label{sec:complexity}

\subsection{Time Complexity}
\begin{itemize}
    \item Direct memory row copy requires $D$ elementary assignments per token lookup.
    \item For batch $B$ and sequence length $S$, time complexity is strictly:
    \begin{equation}
    T_{\text{gather}}(B, S, D) = \mathcal{O}(B \cdot S \cdot D).
    \end{equation}
    (Compared to $\mathcal{O}(B \cdot S \cdot V \cdot D)$ for dense one-hot multiplication).
\end{itemize}

\subsection{Space Complexity}
\begin{itemize}
    \item Output gathered tensor requires $\mathcal{O}(B \cdot S \cdot D)$ space.
    \item Vocabulary table storage is $\mathcal{O}(V \cdot D)$.
\end{itemize}

\section{Worked Example}
\label{sec:worked_example}

Consider vocabulary size $V=4$, embedding dimension $D=3$, batch $B=1$, sequence length $S=2$:
\begin{equation}
\mathbf{E} = \begin{bmatrix}
0.1 & 0.2 & 0.3 \\
1.0 & 0.0 & -1.0 \\
0.5 & 0.5 & 0.5 \\
-2.0 & 1.5 & 0.0
\end{bmatrix}, \quad
\mathbf{I} = [[1, 3]].
\end{equation}

\paragraph{1. Token 1 ($I_{1, 1} = 1$):} Gather row 1 $\implies \mathbf{y}_1 = [1.0, 0.0, -1.0]$.
\paragraph{2. Token 2 ($I_{1, 2} = 3$):} Gather row 3 $\implies \mathbf{y}_2 = [-2.0, 1.5, 0.0]$.

\begin{equation}
\mathbf{Y} = [[[1.0, 0.0, -1.0], \; [-2.0, 1.5, 0.0]]].
\end{equation}

\paragraph{3. Sequence Mean Pooling ($\text{combiner} = \text{mean}$):}
\begin{equation}
\bar{\mathbf{y}} = \frac{1}{2} \left( [1.0, 0.0, -1.0] + [-2.0, 1.5, 0.0] \right) = \frac{1}{2} [-1.0, 1.5, -1.0] = [-0.50, 0.75, -0.50].
\end{equation}
Sparse gradient mask: $\mathbf{m} = [0, 1, 0, 1]^\top$ (active unique indices: $\{1, 3\}$).

\section{Numerical Considerations}
\label{sec:numerical}

\begin{enumerate}
    \item \textbf{Index Bounds Verification:} Attempting to index $k \ge V$ or $k < 0$ causes out-of-bounds segfaults in native kernels; strictly validated in $\mathcal{O}(B \cdot S)$ time before lookup.
    \item \textbf{Deterministic Float Passthrough:} Because lookup is an exact index-gather operation without arithmetic floating-point rounding, forward evaluation is bitwise identical to table memory.
\end{enumerate}
