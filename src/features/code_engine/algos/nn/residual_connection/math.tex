\begin{abstract}
This document establishes the mathematical foundations, gradient flow dynamics, and branch scaling theory of residual (skip) connections in deep neural networks and Transformers. Every concept is formulated across three levels of explanation: intuition, formal formulation, and practical application. We prove the Gradient Highway Unbounded Backpropagation Theorem with explicit separation of assumptions, guarantees, and limitations, derive projection shortcut mechanics and branch variance stabilization rules (DeepNorm / LayerScale), and provide an exact verified numerical trace matching the production implementation contract.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}

Let $\mathbb{R}$ denote the field of real numbers, and $\mathbb{R}^{B \times D}$ the matrix space. Tensors are denoted by uppercase bold letters ($\mathbf{X}, \mathbf{Y}, \mathbf{F}, \mathbf{W}$), vectors by lowercase bold letters ($\mathbf{x}, \mathbf{g}$), and scalar indices by standard italics. The identity operator is denoted by $\mathbf{I}$.

\subsection{Residual Mapping and Additive Identity Shortcut}
\paragraph{Intuition:} Instead of forcing a stack of neural network layers to learn an entire unconstrained mapping from input to output, a residual connection adds the original input directly to the layer's output. The layer only has to learn the differential perturbation (the ``residual'' $\mathcal{F}(\mathbf{x})$), while the un-altered input passes straight through an identity highway.

\paragraph{Mathematical Formulation:}
For an input tensor $\mathbf{X} \in \mathbb{R}^{B \times D}$ and non-linear sublayer transformation $\mathcal{F}(\mathbf{X}) \in \mathbb{R}^{B \times D}$, the standard residual operator is:
\begin{equation}
\label{eq:res_def}
\mathbf{Y} = \mathbf{X} + \alpha \cdot \mathcal{F}(\mathbf{X}),
\end{equation}
where $\alpha \in \mathbb{R}_{>0}$ is an optional branch scaling multiplier (default $\alpha = 1.0$).

\paragraph{Worked Numerical Example:}
Let $\mathbf{x} = [2.0, -1.0]$ and sublayer output $\mathcal{F}(\mathbf{x}) = [0.5, 3.0]$ with $\alpha = 1.0$:
\begin{equation}
\mathbf{y} = \mathbf{x} + 1.0 \cdot \mathcal{F}(\mathbf{x}) = [2.0 + 0.5, -1.0 + 3.0] = [2.5, 2.0].
\end{equation}

\subsection{Summary of Symbols}
\begin{itemize}
    \item $B \in \mathbb{N}$: Batch size dimension.
    \item $D_{\text{in}} \in \mathbb{N}$: Input identity channel dimension.
    \item $D_{\text{out}} \in \mathbb{N}$: Sublayer output channel dimension.
    \item $\mathbf{X} \in \mathbb{R}^{B \times D_{\text{in}}}$: Input identity/skip tensor.
    \item $\mathbf{F} \in \mathbb{R}^{B \times D_{\text{out}}}$: Sublayer transformation tensor $\mathcal{F}(\mathbf{X})$.
    \item $\mathbf{W}_{\text{proj}} \in \mathbb{R}^{D_{\text{out}} \times D_{\text{in}}}$: Optional projection shortcut matrix for dimension matching.
    \item $\alpha \in \mathbb{R}_{>0}$: Residual branch scaling factor.
    \item $\mathbf{g} \in [0, 1]^{D_{\text{out}}}$: Learned gating parameter vector in gated residuals.
    \item $\mathbf{Y} \in \mathbb{R}^{B \times D_{\text{out}}}$: Output combined tensor.
\end{itemize}

\section{Problem Definition and Core Variants}
\label{sec:problem_definition}

\begin{definition}[Standard and Scaled Residual Connections]
When $D_{\text{in}} = D_{\text{out}}$:
\begin{equation}
\mathbf{Y} = \mathbf{X} + \alpha \cdot \mathbf{F}.
\end{equation}
In deep networks ($L > 100$), setting $\alpha = \frac{1}{\sqrt{2}}$ or LayerScale $\alpha \ll 1$ prevents the variance of $\mathbf{Y}$ from exploding exponentially across layers.
\end{definition}

\begin{definition}[Linear Projection Shortcut (He et al. 2016)]
When $D_{\text{in}} \ne D_{\text{out}}$, the skip path is projected through linear matrix $\mathbf{W}_{\text{proj}}$:
\begin{equation}
\label{eq:proj_shortcut}
\mathbf{Y} = \mathbf{X} \mathbf{W}_{\text{proj}}^\top + \alpha \cdot \mathbf{F} \in \mathbb{R}^{B \times D_{\text{out}}}.
\end{equation}
\end{definition}

\begin{definition}[Gated Residual Connection (Highway Network / GRU)]
Using a channel-wise learned gate vector $\mathbf{g} \in \mathbb{R}^{D_{\text{out}}}$ with sigmoid activation $\sigma(\mathbf{g})$:
\begin{equation}
\label{eq:gated_res}
\mathbf{Y} = \sigma(\mathbf{g}) \odot \mathbf{X} + (1 - \sigma(\mathbf{g})) \odot \mathbf{F}.
\end{equation}
\end{definition}

\section{Algorithmic Specification}
\label{sec:algorithm}

Algorithm~\ref{alg:residual} defines the deterministic forward execution implemented in \texttt{impl.py}.

\begin{algorithm}[H]
\caption{Deterministic Residual Combination and Projection Execution}
\label{alg:residual}
\begin{algorithmic}[1]
\Require Identity tensor $\mathbf{X} \in \mathbb{R}^{B \times D_{\text{in}}}$, sublayer tensor $\mathbf{F} \in \mathbb{R}^{B \times D_{\text{out}}}$, projection matrix $\mathbf{W}_{\text{proj}}$, scale $\alpha$, topology $\in \{\text{standard\_add}, \text{scaled\_add}, \text{gated\_residual}\}$, gate vector $\mathbf{g}$.
\Ensure Output tensor $\mathbf{Y} \in \mathbb{R}^{B \times D_{\text{out}}}$, cosine similarity $s_{\text{cos}}$, energy ratio $\rho_{\text{res}}$.
\State $B \gets \text{rows}(\mathbf{X}), \quad D_{\text{in}} \gets \text{cols}(\mathbf{X}), \quad D_{\text{out}} \gets \text{cols}(\mathbf{F})$
\If{$D_{\text{in}} \ne D_{\text{out}}$}
    \State $\mathbf{X}_{\text{skip}} \gets \mathbf{X} \mathbf{W}_{\text{proj}}^\top \in \mathbb{R}^{B \times D_{\text{out}}}$
\Else
    \State $\mathbf{X}_{\text{skip}} \gets \mathbf{X}$
\EndIf
\State $\mathbf{Y} \gets \mathbf{0} \in \mathbb{R}^{B \times D_{\text{out}}}$
\For{$i = 1 \textbf{ to } B$}
    \For{$j = 1 \textbf{ to } D_{\text{out}}$}
        \State $x \gets X_{\text{skip}, ij}, \quad f \gets F_{ij}$
        \If{$\text{topology} = \text{gated\_residual}$}
            \State $gate \gets 1.0 / (1.0 + \exp(-g_j))$
            \State $Y_{ij} \gets gate \cdot x + (1.0 - gate) \cdot f$
        \Else
            \State $Y_{ij} \gets x + \alpha \cdot f$
        \EndIf
    \EndFor
\EndFor
\State Compute cosine similarity $s_{\text{cos}} = \frac{\langle \mathbf{X}_{\text{skip}}, \mathbf{Y} \rangle}{\|\mathbf{X}_{\text{skip}}\|_F \|\mathbf{Y}\|_F}$ and residual ratio $\rho_{\text{res}} = \frac{\|\alpha \mathbf{F}\|_F}{\|\mathbf{X}_{\text{skip}}\|_F}$
\State \Return $(\mathbf{Y}, s_{\text{cos}}, \rho_{\text{res}})$
\end{algorithmic}
\end{algorithm}

\section{Theoretical Guarantees: Gradient Highway Theorem}
\label{sec:correctness}

\begin{theorem}[Gradient Highway Unbounded Flow (He et al. 2016)]
\label{thm:gradient_highway}
\textbf{Assumptions:}
Consider an $L$-layer residual network where every block uses an identity skip connection:
\begin{equation}
\mathbf{x}_{l+1} = \mathbf{x}_l + \mathcal{F}_l(\mathbf{x}_l), \quad \forall l \in \{0, \dots, L-1\}.
\end{equation}

\textbf{Guarantee:}
Any state $\mathbf{x}_L$ at depth $L$ can be expanded additively as:
\begin{equation}
\label{eq:additive_expansion}
\mathbf{x}_L = \mathbf{x}_l + \sum_{i=l}^{L-1} \mathcal{F}_i(\mathbf{x}_i).
\end{equation}
By the chain rule of differentiation, the gradient of loss $\mathcal{L}$ with respect to layer $l$ is:
\begin{equation}
\label{eq:highway_grad}
\frac{\partial \mathcal{L}}{\partial \mathbf{x}_l} = \frac{\partial \mathcal{L}}{\partial \mathbf{x}_L} \frac{\partial \mathbf{x}_L}{\partial \mathbf{x}_l} = \frac{\partial \mathcal{L}}{\partial \mathbf{x}_L} \left( \mathbf{I} + \frac{\partial}{\partial \mathbf{x}_l} \sum_{i=l}^{L-1} \mathcal{F}_i(\mathbf{x}_i) \right).
\end{equation}
The gradient contains the additive identity term $\frac{\partial \mathcal{L}}{\partial \mathbf{x}_L} \cdot \mathbf{I}$, guaranteeing that the gradient flows directly from output layer $L$ to initial layer $l$ without exponentially vanishing, even when the sublayer Jacobian $\frac{\partial \mathcal{F}_i}{\partial \mathbf{x}_l} \approx \mathbf{0}$.

\textbf{Limitations:}
The identity gradient flow breaks if non-linear activations (such as post-activation ReLU) or non-identity scaling are placed directly on the skip path itself.
\end{theorem}

\section{Complexity Derivation}
\label{sec:complexity}

\subsection{Time Complexity}
\begin{enumerate}
    \item When $D_{\text{in}} = D_{\text{out}}$, addition $\mathbf{X} + \alpha \mathbf{F}$ takes exactly $B \cdot D$ scalar additions and multiplications ($\mathcal{O}(B \cdot D)$).
    \item When projection is required ($D_{\text{in}} \ne D_{\text{out}}$), $\mathbf{X} \mathbf{W}_{\text{proj}}^\top$ takes $\mathcal{O}(B \cdot D_{\text{in}} \cdot D_{\text{out}})$ multiply-adds.
    \item Total time complexity:
    \begin{equation}
    T(B, D_{\text{in}}, D_{\text{out}}) = \mathcal{O}(B \cdot D_{\text{out}} + B \cdot D_{\text{in}} \cdot D_{\text{out}} \cdot \mathbf{1}_{\{D_{\text{in}} \ne D_{\text{out}}\}}).
    \end{equation}
\end{enumerate}

\subsection{Space Complexity}
\begin{enumerate}
    \item Output tensor $\mathbf{Y}$ allocates $B \cdot D_{\text{out}}$ floating-point numbers.
    \item Space complexity: $\mathcal{O}(B \cdot D_{\text{out}})$.
\end{enumerate}

\section{Worked Example and Numerical Trace}
\label{sec:worked_example}

Let $B = 2, D_{\text{in}} = 2, D_{\text{out}} = 2$, $\alpha = 0.5$.
\begin{equation}
\mathbf{X} = \begin{bmatrix} 1.0 & 2.0 \\ -1.0 & 0.0 \end{bmatrix}, \quad
\mathbf{F} = \begin{bmatrix} 0.4 & -0.8 \\ 2.0 & 1.0 \end{bmatrix}.
\end{equation}
Topology: \texttt{scaled\_add}.

\subsection{Exact Element Calculations}
\begin{align}
Y_{11} &= X_{11} + \alpha F_{11} = 1.0 + 0.5(0.4) = 1.0 + 0.2 = 1.2, \\
Y_{12} &= X_{12} + \alpha F_{12} = 2.0 + 0.5(-0.8) = 2.0 - 0.4 = 1.6, \\
Y_{21} &= X_{21} + \alpha F_{21} = -1.0 + 0.5(2.0) = -1.0 + 1.0 = 0.0, \\
Y_{22} &= X_{22} + \alpha F_{22} = 0.0 + 0.5(1.0) = 0.0 + 0.5 = 0.5.
\end{align}
Output matrix:
\begin{equation}
\mathbf{Y} = \begin{bmatrix} 1.2 & 1.6 \\ 0.0 & 0.5 \end{bmatrix}.
\end{equation}
